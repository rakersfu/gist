# 管理ddns-updater
修改 systemd 服务文件
nano /etc/systemd/system/ddns-updater.service
将
StandardOutput=append:/var/log/ddns-updater/output.log
StandardError=append:/var/log/ddns-updater/error.log
替换为
StandardOutput=journal
StandardError=journal
重新加载 systemd 并重启服务
systemctl daemon-reload
systemctl restart ddns-updater
查看日志
journalctl -u ddns-updater -f  # 实时查看
journalctl -u ddns-updater --since "1 hour ago"  # 查看最近1小时
查看状态
systemctl status ddns-updater
