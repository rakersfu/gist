logrotate -df /etc/logrotate.d/nginx-custom 2>&1 | grep -E '^running postrotate script'

# 3) File check: should report "ASCII text", no CRLF, no tabs
file /etc/logrotate.d/nginx-custom
od -c /etc/logrotate.d/nginx-custom | head -5
grep -c $'\r' /etc/logrotate.d/nginx-custom
grep -Pn '\t' /etc/logrotate.d/nginx-custom
cat -A /etc/logrotate.d/nginx-custom | grep -nE '\^I|\^M'
